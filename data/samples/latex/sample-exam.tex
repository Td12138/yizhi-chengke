\section*{二、计算题}

\begin{enumerate}[resume]
  \item 计算定积分 $\displaystyle \int_{0}^{1} x e^{x}\,dx$。

  \item 判断级数 $\displaystyle \sum_{n=1}^{\infty}\frac{1}{n^{2}+n}$ 是否收敛，并求其和。
\end{enumerate}

\section*{三、证明题}

\begin{enumerate}[resume]
  \item 设 $f$ 在 $[0,1]$ 上连续，在 $(0,1)$ 内可导，且 $f(0)=f(1)=0$。
        证明：存在 $\xi \in (0,1)$，使得 $f'(\xi)=f(\xi)$。（提示：考虑 $g(x)=e^{-x}f(x)$。）
\end{enumerate}
